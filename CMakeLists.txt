cmake_minimum_required(VERSION 3.16)
project(lifecase VERSION 0.1.0 LANGUAGES CXX)

add_executable(lifecase-fixture fixture/main.cpp)
target_compile_features(lifecase-fixture PRIVATE cxx_std_17)
set_target_properties(lifecase-fixture PROPERTIES CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
target_link_libraries(lifecase-fixture PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(lifecase-fixture PRIVATE /W4 /WX /utf-8 /permissive-)
  target_compile_definitions(lifecase-fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
else()
  target_compile_options(lifecase-fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MINGW)
  target_link_options(lifecase-fixture PRIVATE -municode)
endif()

include(GNUInstallDirs)
install(TARGETS lifecase-fixture RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME native-fixture
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/fixture/tests.py $<TARGET_FILE:lifecase-fixture>)
    set_tests_properties(native-fixture PROPERTIES TIMEOUT 60)
  endif()
endif()
